🌱 Beginner — what is prom-client?
prom-client is the Node.js library that collects metrics (counts, timings) and exposes them at a URL endpoint. Prometheus — an open-source monitoring server — reads that endpoint. We use prom-client directly without needing a full Prometheus server.

config/metrics.js
const promClient = require('prom-client');

// Create a registry — container for all our metrics
const register = new promClient.Registry();

// Collect default Node.js metrics (CPU, memory, event loop lag)
promClient.collectDefaultMetrics({
  register,
  prefix: 'cloudeats_user_'
});

// ── COUNTER: total requests (only ever increases) ──
const httpRequestsTotal = new promClient.Counter({
  name: 'http_requests_total',
  help: 'Total number of HTTP requests received',
  labelNames: ['method', 'route', 'status_code'],
  registers: [register]
});

// ── GAUGE: active requests right now (can go up or down) ──
const activeRequests = new promClient.Gauge({
  name: 'http_requests_active',
  help: 'Number of HTTP requests currently being processed',
  registers: [register]
});

// ── HISTOGRAM: response time distribution ──
const httpRequestDuration = new promClient.Histogram({
  name: 'http_request_duration_seconds',
  help: 'HTTP request duration in seconds',
  labelNames: ['method', 'route', 'status_code'],
  buckets: [0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0],
  registers: [register]
});

// ── BUSINESS METRICS ──
const userRegistrationsTotal = new promClient.Counter({
  name: 'user_registrations_total',
  help: 'Total user registrations',
  registers: [register]
});

const userLoginsTotal = new promClient.Counter({
  name: 'user_logins_total',
  help: 'Total login attempts',
  labelNames: ['status'],   // 'success' or 'failure'
  registers: [register]
});

// ── MIDDLEWARE: called on every request ──
function metricsMiddleware(req, res, next) {
  activeRequests.inc();
  const startTime = Date.now();

  res.on('finish', () => {
    activeRequests.dec();
    const duration = (Date.now() - startTime) / 1000;
    const route = req.route ? req.route.path : req.path;

    httpRequestsTotal.inc({
      method: req.method,
      route,
      status_code: res.statusCode
    });

    httpRequestDuration.observe(
      { method: req.method, route, status_code: res.statusCode },
      duration
    );
  });

  next();
}

// ── HELPER FUNCTIONS for business events ──
const trackRegistration = () => userRegistrationsTotal.inc();
const trackLogin = (success) => userLoginsTotal.inc({ status: success ? 'success' : 'failure' });

// ── EXPORT metrics text (Prometheus format) ──
const getMetrics = () => register.metrics();

// ── EXPORT metrics as JSON (for our dashboard) ──
const getMetricsJSON = () => register.getMetricsAsJSON();

module.exports = {
  metricsMiddleware,
  trackRegistration,
  trackLogin,
  getMetrics,
  getMetricsJSON
};
🌱 Beginner — three metric types explained
Counter: like an odometer — only goes up. Total requests ever made.
Gauge: like a fuel gauge — can go up or down. Active requests right now.
Histogram: like a speed distribution chart — counts how many requests fell into each time bucket (under 10ms, under 50ms, under 100ms, etc.).
⚡ Average — labels
Labels let you slice metrics by dimension. httpRequestsTotal{method="POST", route="/api/auth/login", status_code="401"} tracks failed login requests specifically. Without labels, you only know total requests — with labels, you can ask "how many POST requests to /api/auth/login returned 401 in the last minute?"
Part 3 Add Middleware and /metrics Endpoints ⏱ 10 min
3
Import metrics in server.js and register middleware

Add the import line near the top of server.js, then add the middleware early in the middleware chain:

server.js — imports and middleware
// Add with other requires:
const { metricsMiddleware, trackRegistration, trackLogin, getMetrics, getMetricsJSON } = require('./config/metrics');

// Add to middleware section BEFORE your routes (after cors and json):
app.use(cors());
app.use(express.json());
app.use(morganMiddleware);
app.use(metricsMiddleware);    // ← ADD THIS — must be before routes
⚠️
metricsMiddleware must be registered with app.use() before any route definitions. If it comes after, some routes will not have their metrics recorded.
4
Add the /metrics and /metrics/json endpoints

Add these two routes in server.js — they serve the metrics data. Place them with your other route definitions:

server.js — metrics endpoints
// ── METRICS ENDPOINTS ──

// Prometheus text format (what Prometheus server would scrape)
app.get('/metrics', async (req, res) => {
  res.set('Content-Type', 'text/plain; version=0.0.4');
  res.send(await getMetrics());
});

// JSON format (what our Chart.js dashboard uses)
app.get('/metrics/json', async (req, res) => {
  res.json(await getMetricsJSON());
});
🚀 Advanced — secure the metrics endpoint
In production, the /metrics endpoint should NOT be public — it exposes internal system details. Options: (1) restrict it to internal network only using a security group rule, (2) add basic auth: app.get('/metrics', authenticateToken, requireRole('admin'), ...), (3) run it on a separate internal port that is not exposed outside the Docker network.
Part 4 Track Business Events ⏱ 8 min
5
Call trackLogin and trackRegistration at event points

In your login endpoint, call trackLogin() just before the response. In your registration endpoint, call trackRegistration() on success:

server.js — login endpoint update
// Inside login — password verified:
trackLogin(true);    // ← ADD before returning token
logger.info('User login successful', { userId: user.id, email });
return res.json({ token, user: { ... } });

// Inside login — password wrong:
trackLogin(false);   // ← ADD before returning 401
logger.warn('Login failed - invalid password', { email, ip: req.ip });
return res.status(401).json({ error: 'Invalid email or password' });

// Inside registration — INSERT succeeded:
trackRegistration(); // ← ADD after successful insert
logger.info('New user registered', { userId: result.insertId, email });
Why track business metrics separately? The metricsMiddleware tracks all HTTP requests. But it cannot tell the difference between a successful login and a failed login from the route alone — both go to POST /api/auth/login. Business metrics let you slice by outcome, not just by route.

<script>
let statusChart, durationChart;

async function loadMetrics() {
  try {
    const resp = await fetch('/metrics/json');
    const data = await resp.json();
    document.getElementById('lastUpdated').textContent = 'Updated: ' + new Date().toLocaleTimeString();

    let totalReq = 0, successReq = 0, errorReq = 0;
    let loginSuccess = 0, loginFail = 0, registrations = 0;
    const statusMap = {}, durationBuckets = {};

    data.forEach(metric => {
      if (metric.name === 'http_requests_total') {
        metric.values.forEach(v => {
          totalReq += v.value;
          const sc = v.labels.status_code;
          if (sc < 400) successReq += v.value; else errorReq += v.value;
          statusMap[sc] = (statusMap[sc] || 0) + v.value;
        });
      }
      if (metric.name === 'http_requests_active') {
        document.getElementById('activeReq').textContent = metric.values[0]?.value ?? 0;
      }
      if (metric.name === 'user_logins_total') {
        metric.values.forEach(v => {
          if (v.labels.status === 'success') loginSuccess += v.value;
          else loginFail += v.value;
        });
      }
      if (metric.name === 'user_registrations_total') {
        registrations = metric.values[0]?.value ?? 0;
      }
      if (metric.name === 'http_request_duration_seconds') {
        metric.values.forEach(v => {
          if (v.labels.le && v.labels.le !== '+Inf') {
            const le = v.labels.le;
            durationBuckets[le] = (durationBuckets[le] || 0) + v.value;
          }
        });
      }
    });

    document.getElementById('totalRequests').textContent = totalReq;
    document.getElementById('successRate').textContent = totalReq > 0 ? Math.round(successReq/totalReq*100)+'%' : '—';
    document.getElementById('logins').textContent = loginSuccess;
    document.getElementById('failedLogins').textContent = loginFail;
    document.getElementById('registrations').textContent = registrations;

    // Status breakdown bar chart
    const statusLabels = Object.keys(statusMap).sort();
    const statusData = statusLabels.map(k => statusMap[k]);
    const statusColors = statusLabels.map(k => k < 400 ? '#00e5a0' : k < 500 ? '#ffb740' : '#ff5e7a');
    if (statusChart) statusChart.destroy();
    statusChart = new Chart(document.getElementById('statusChart'), {
      type: 'bar',
      data: { labels: statusLabels, datasets: [{ label: 'Requests', data: statusData, backgroundColor: statusColors }] },
      options: { plugins: { legend: { display: false } }, scales: { x: { ticks: { color: '#6b7fa0' } }, y: { ticks: { color: '#6b7fa0' } } } }
    });

    // Duration histogram bar chart
    const bucketLabels = Object.keys(durationBuckets).sort((a,b) => parseFloat(a)-parseFloat(b));
    const bucketData = bucketLabels.map(k => durationBuckets[k]);
    if (durationChart) durationChart.destroy();
    durationChart = new Chart(document.getElementById('durationChart'), {
      type: 'bar',
      data: { labels: bucketLabels.map(l => '≤'+l+'s'), datasets: [{ label: 'Count', data: bucketData, backgroundColor: '#00c8ff' }] },
      options: { plugins: { legend: { display: false } }, scales: { x: { ticks: { color: '#6b7fa0' } }, y: { ticks: { color: '#6b7fa0' } } } }
    });

  } catch(e) {
    console.error('Could not load metrics', e);
  }
}

loadMetrics();

</script>

🚀 Advanced Challenge Extension

The Prometheus ecosystem is the industry standard for metrics. Set up a minimal Prometheus + Grafana stack as an optional extra:

  1. Add a prometheus service to docker-compose.yml
  2. Create a prometheus.yml config that scrapes user-service:3001/metrics every 15 seconds
  3. Add a grafana service (port 3000), connect it to Prometheus as a data source
  4. Import the "Node.js" dashboard template from grafana.com/grafana/dashboards

This is the full observability stack used at startups and mid-sized companies. Your prom-client code works unchanged — you just add the server-side infrastructure.
